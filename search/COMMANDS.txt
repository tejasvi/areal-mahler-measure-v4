  OMP_NUM_THREADS=16 nice -n 10 $PY -I arl_primal.py A58 uniform:20:0.01 L1 > primal_L1.log            -> primal_L1.npy
  TP_NPROC=40 nice -n 10 $PY -I arl_dual.py primal_L1.npy A58 > arldual_L1.log                         -> primal_L1_arldualN_A58.npy, _idx.json
  nice -n 10 $PY -I export_arl_lo.py primal_L1.npy primal_L1_arldualN_A58.npy A58 3e-6 cert_arllo_L1.json
